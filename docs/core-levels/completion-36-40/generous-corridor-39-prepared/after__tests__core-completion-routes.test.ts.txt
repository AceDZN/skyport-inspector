const heldTraceCounts=new WeakMap<CompletionRoute,number>();
function heldContacts(r:CompletionRoute,label:string){const id=r.w.heldId;if(!id)return;const count=heldTraceCounts.get(r)??0;if(count>=96)return;const controller=r.w.holdController,hits=Array.from({length:controller.numComputedCollisions()},(_,i)=>controller.computedCollision(i)).filter(Boolean).map(c=>({id:c!.collider?(r.w.fixed.get(c!.collider!.handle)?.id??r.w.crates.find(x=>x.collider.handle===c!.collider!.handle)?.id??'native-machine'):null,normal:c!.normal1,witness:c!.witness1,applied:c!.translationDeltaApplied,remaining:c!.translationDeltaRemaining}));if(!hits.length)return;heldTraceCounts.set(r,count+1);r.checkpoints.push({label:'read-only-held-contact-'+label,time:r.w.time,player:r.w.player.pos.toArray(),cube:r.cube(id).pos.toArray(),carryTarget:r.w.carryTarget().toArray(),hits});}
/** Normal camera pitch raises held cargo over upcoming real stair treads. */
function carryWalk(r:CompletionRoute,x:number,z:number,pitch=.18){assert.ok(r.w.heldId);for(let i=0;i<2400;i++){const dx=x-r.w.player.pos.x,dz=z-r.w.player.pos.z;if(Math.hypot(dx,dz)<.14){r.tick(.1);heldContacts(r,'waypoint');return;}r.w.yaw=Math.atan2(-dx,-dz);r.w.pitch=pitch;r.step(forward);if(i%12===0)heldContacts(r,'walk');}assert.fail('Ordinary raised-cargo WASD blocked '+r.detail());}

}
