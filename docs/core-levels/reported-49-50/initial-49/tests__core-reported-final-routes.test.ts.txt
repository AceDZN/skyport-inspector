/** Dedicated unregistered49 native routes; neither browser nor progression proof.
 * No actor/cargo/velocity/machine/history/goal/save setters or reset calls.
 */
import test,{before}from'node:test';import assert from'node:assert/strict';
import{initializePhysics}from'../src/core/world';import{V}from'../src/physics';
import{ReportedFinalRoute,route}from'./helpers/reported-final-route';
if(process.env.RIFT_REPORTED_FINAL_ROUTES==='1'){
before(initializePhysics);
function journal(r:ReportedFinalRoute){const j=r.w.snapshot().journal;assert.ok(j);return j;}
function pair(r:ReportedFinalRoute,upper:boolean){
 r.walk(8.5,12.5);r.shot(0,'copy-source',V(10.985,1.55,8));if(!upper)r.walk(-1,12.5);r.shot(1,upper?'gallery-return':'loading-return',V(upper?5:-5,upper?4.95:1.55,-13.985));
 assert.equal(r.pad('loading-pressure').pressed,false);assert.equal(r.pad('gallery-pressure').pressed,false);assert.equal(r.w.machines.flags['loading-light'],false);assert.equal(r.w.machines.flags['gallery-light'],false);r.note('actual-placed-link-is-not-supplied');
}
function sourceCargo(r:ReportedFinalRoute){r.walk(-6,12.5);r.walk(-6,8);r.pickup('copy-cube');r.walk(-6,10.5);}
function transferToStation(r:ReportedFinalRoute,upper:boolean){
 assert.equal(r.w.heldId,'copy-cube');r.walk(8.5,10.5);r.walk(9.5,8);const before=r.w.player.crossings;r.enter(0);assert.equal(r.w.player.crossings,before+1);assert.ok(r.cube('copy-cube').crossings>=1,'A held flag cannot replace real cargo transport');
 const x=upper?8:-8,y=upper?3.4:0,id=upper?'gallery':'loading';r.walk(upper?7.5:-7.5,-12);r.dropPad('copy-cube',id+'-pressure',x,-12.8);
 assert.ok(r.w.player.grounded&&Math.abs(r.w.player.pos.y-(y+.865))<.06,'Actual station needs its authored floor height');assert.equal(r.pad(id+'-pressure').crateId,'copy-cube');assert.equal(r.pad((upper?'loading':'gallery')+'-pressure').pressed,false);
 r.until(()=>r.w.machines.flags[id+'-light']===true&&journal(r).now.some(x=>x.id===id+'-light'&&x.state==='LIT'),2,'Current local pressure and real optical link did not supply the named receiver');assert.equal(r.w.machines.flags[(upper?'loading':'gallery')+'-light'],false);r.note('actual-local-cargo-and-optical-supply-'+id);
}
function passage(r:ReportedFinalRoute,upper:boolean){const id=upper?'gallery':'loading',x=upper?9.5:-9.5;r.walk(x,-12.8);r.walk(x,-10);r.until(()=>journal(r).now.some(x=>x.id===id+'-door'&&x.state==='OPEN'),2,'Local current pressure/light AND leaf never fully clears');r.walk(upper?14:-14,-10);}
function finish(r:ReportedFinalRoute,upper:boolean){passage(r,upper);r.finish(upper?'gallery-exit':'loading-exit');}
test('native C49 both reports retain identical fixed originals, both copies and physical local circuits',()=>{const a=new ReportedFinalRoute(49),b=new ReportedFinalRoute(49,true);try{assert.deepEqual(a.w.blueprint,b.w.blueprint);assert.deepEqual(a.w.level.optics,b.w.level.optics);assert.deepEqual(journal(a).rows,[]);for(const text of['LOADING told the operator','to leave the cube THERE','GALLERY told the operator','to leave the cube HERE'])assert.ok(a.w.blueprint.marks.some(x=>x.text===text),'Both actual copies must exist without selected-mission geometry');assert.equal(a.pad('loading-pressure').pressed,false);assert.equal(a.pad('gallery-pressure').pressed,false);assert.equal(a.w.machines.flags['loading-light'],false);assert.equal(a.w.machines.flags['gallery-light'],false);}finally{a.w.dispose();b.w.dispose();}});
test('native C49 copied LOADING instruction supplies the lower station and its own receiver',()=>route(49,false,'loading-there-local-supply',r=>{pair(r,false);sourceCargo(r);transferToStation(r,false);finish(r,false);}));
test('native C49 same-place GALLERY report carries real cargo to the upper local supply',()=>route(49,true,'gallery-here-upper-supply',r=>{pair(r,true);sourceCargo(r);transferToStation(r,true);finish(r,true);}));
test('native C49 mixed source allocation physically closes both exits until the real optical return is corrected',()=>route(49,false,'mixed-source-relink',r=>{pair(r,false);sourceCargo(r);transferToStation(r,false);r.shot(1,'gallery-return',V(5,4.95,-13.985));r.until(()=>r.w.machines.flags['gallery-light']===true&&!r.w.machines.flags['loading-light'],2,'Mixed actual link did not light the other station');r.until(()=>journal(r).now.some(x=>x.id==='loading-door'&&x.state==='CLOSED'),2,'LOADING leaf ignored its missing own receiver');assert.ok(r.device('gallery-door').progress<.001,'GALLERY leaf ignored its missing own pressure');assert.equal(r.pad('loading-pressure').crateId,'copy-cube');assert.equal(r.pad('gallery-pressure').pressed,false);r.note('actual-mixed-source-neither-local-AND-complete');r.shot(1,'loading-return',V(-5,1.55,-13.985));r.until(()=>r.w.machines.flags['loading-light']===true,2,'Corrected local receiver failed');finish(r,false);}));
test('native C49 wrong complete GALLERY branch can return the same cargo and fulfill the copied LOADING source',()=>route(49,false,'wrong-gallery-return-loading',r=>{pair(r,true);sourceCargo(r);transferToStation(r,true);passage(r,true);assert.equal(r.w.campaignResult,undefined);assert.ok(r.events.some(e=>e.type==='wrong-exit'&&e.exit==='gallery-exit'));r.walk(9.5,-10);r.walk(9.5,-12.8);r.walk(8,-12.8);r.pickup('copy-cube');r.walk(7.5,-12.8);r.walk(5,-12.5);r.enter(1);assert.equal(r.w.heldId,'copy-cube');assert.ok(r.cube('copy-cube').pos.y<3,'Actual cargo did not return from upper station');r.walk(9.5,10.5);r.walk(-1,12.5);r.shot(1,'loading-return',V(-5,1.55,-13.985));transferToStation(r,false);finish(r,false);}));
test('native C49 actual cargo pressure loss extinguishes the local receiver and replacement restores its route',()=>route(49,false,'local-current-loss-restore',r=>{pair(r,false);sourceCargo(r);transferToStation(r,false);r.pickup('copy-cube');r.w.pitch=.18;r.tick(.4);r.walk(-9.5,-12.8);r.until(()=>!r.pad('loading-pressure').pressed&&!r.w.machines.flags['loading-light'],2,'Actual supply removal failed to extinguish local receiver');r.until(()=>journal(r).now.some(x=>x.id==='loading-door'&&x.state==='CLOSED'),2,'Current supply loss failed to close actual local leaf');assert.equal(r.w.campaignResult,undefined);assert.ok(journal(r).now.some(x=>x.id==='loading-pressure'&&x.state==='EMPTY'));r.note('actual-previous-report-does-not-maintain-empty-current-supply');r.dropPad('copy-cube','loading-pressure',-8,-12.8);r.until(()=>r.w.machines.flags['loading-light']===true,2,'Same-cargo replacement failed actual supply');finish(r,false);}));
}else test('unregistered final reported drafts await explicit native acceptance',{skip:'Enable RIFT_REPORTED_FINAL_ROUTES=1 only in the assigned serial slot.'},()=>{});
