/** Source-only prepared C40 input routes. Execute only in an explicit serial grant.
 * No actor/cargo/velocity/machine/history/save setters or reset calls.
 */
import test,{before}from'node:test';import assert from'node:assert/strict';
import{initializePhysics}from'../src/core/world';import{V}from'../src/physics';
import{CompletionRoute,route,forward}from'./helpers/completion-route';
if(process.env.RIFT_COMPLETION_ROUTES==='1'){
before(initializePhysics);
function journal(r:CompletionRoute){const j=r.w.snapshot().journal;assert.ok(j);return j;}
function report(r:CompletionRoute,id:string){const x=journal(r).reports.find(x=>x.id===id);assert.ok(x);return x;}
function prepareDeep(r:CompletionRoute){r.walk(-1,7.3);r.walk(3,7.3);r.walk(3,6.45);r.shot(0,'vault-deep-floor',V(3,-17.975,-1.1));r.shot(1,'vault-launch',V(-7.985,9,-6));}
function vaultLanding(r:CompletionRoute){
 prepareDeep(r);r.w.yaw=0;r.w.pitch=0;const crossings=r.w.player.crossings,start=r.events.length;
 for(let i=0;i<480&&r.w.player.crossings===crossings;i++)r.step(forward);
 assert.equal(r.w.player.crossings,crossings+1,'Real deep fall failed first portal crossing '+r.detail());assert.equal(r.events.slice(start).some(e=>e.type==='land'),false,'Bottom support before entry cannot count as the actual deep drop');
 const entry=r.events.filter(e=>e.type==='enter'&&e.body==='player').at(-1);assert.ok(entry);assert.ok(Math.abs(Number(entry.speed)-Number(entry.exitSpeed))<1e-6,'Fling altered speed magnitude');assert.equal(r.w.aperture(r.w.player,r.w.portals[1]!),true,'Actual transformed upright capsule must fit wall exit');
 const samples:unknown[]=[];for(let i=0;i<960&&!r.w.player.grounded;i++){if(i%12===0)samples.push({time:r.w.time,p:r.w.player.pos.toArray(),v:r.w.player.vel.toArray()});r.step();}
 const floor=r.w.solids.find(s=>s.id==='retrieval-vault');assert.ok(floor);const p=r.w.player;assert.ok(p.grounded&&Math.abs(p.pos.y-5.865)<.06,'Actual flight did not land on the high retrieval floor '+r.detail());assert.ok(p.pos.x-p.half.x>=floor.min.x-.015&&p.pos.x+p.half.x<=floor.max.x+.015&&p.pos.z-p.half.z>=floor.min.z-.015&&p.pos.z+p.half.z<=floor.max.z+.015,'Required whole supported high landing');
 assert.equal(r.w.respawns,0);assert.equal(r.pad('gallery-pressure').pressed,false);assert.equal(r.pad('service-pressure').pressed,false);assert.equal(r.device('final-lift').moving,false);assert.ok(r.cube('vault-cube').pos.y>5);assert.equal(report(r,'gallery-contact').status,'NOT YET OBSERVED');r.checkpoints.push({label:'actual-fling-samples',entry,samples});r.note('actual-high-retrieval-is-not-final-delivery');
}
function returnVaultCargo(r:CompletionRoute){
 r.walk(16.5,-6);r.shot(0,'vault-return-wall',V(1,1.55,13.985));r.shot(1,'vault-return-floor',V(18,5.025,-8));r.walk(24.2,-6);r.pickup('vault-cube');r.walk(20,-6);r.walk(20,-8);
 // Ordinary low release over the actual floor aperture. A lower WALL return
 // gives the loose body a real receiving floor rather than a floor/floor cycle.
 r.w.yaw=Math.PI/2;r.w.pitch=-.5;r.tick(.5);r.e();assert.equal(r.w.heldId,undefined);const c=r.cube('vault-cube');
 r.until(()=>c.crossings>=1&&c.pos.y<3&&c.pos.z>8,6,'Actual cargo never crossed from high floor to lower return');r.until(()=>c.pos.y>.45&&c.pos.y<1.1&&c.pos.x>-3&&c.pos.x<8&&c.pos.z>8&&c.pos.z<14&&c.vel.length()<.25,6,'Returned cargo never settles on actual approach floor');r.note('actual-independent-cargo-return-no-pose-setter');
 const before=r.w.player.crossings;r.w.yaw=Math.PI/2;r.w.pitch=0;for(let i=0;i<480&&r.w.player.crossings===before;i++)r.step(forward);assert.equal(r.w.player.crossings,before+1,'Actual operator floor return failed '+r.detail());r.until(()=>r.w.player.grounded&&Math.abs(r.w.player.pos.y-.865)<.08,5,'Operator return did not reach actual lower approach');assert.equal(r.w.respawns,0);
 // The returned body may have rotated or rolled. Approach its CURRENT pose
 // from the east, clear of the recovery opening, and stop before contact.
 r.walk(3.5,13);const current=c.pos.clone();r.walk(current.x+1.8,current.z);r.pickup('vault-cube');r.note('actual-returned-cargo-regrasped-at-current-pose');
}
function allocate40(r:CompletionRoute,pad:'gallery-pressure'|'service-pressure'){
 if(pad==='gallery-pressure'){r.walk(3,12);r.walk(3,9);r.dropPad('vault-cube',pad,1.8,9);}else{r.walk(3,12);r.dropPad('vault-cube',pad,4.2,12);}assert.equal(r.pad(pad).crateId,'vault-cube');
}
function finalGallery(r:CompletionRoute){
 const lift=r.device('final-lift');r.walk(3,9);r.walk(3,6.8);r.walk(-1,6.2);r.until(()=>!lift.moving&&Math.abs(lift.position.y+.2)<.015&&journal(r).now.some(x=>x.id==='final-lift'&&x.state==='AT LOADING'),35,'Real final transfer did not stop at LOADING');r.walk(-1,3);
 r.until(()=>!lift.moving&&Math.abs(lift.position.y-3.2)<.015&&journal(r).now.some(x=>x.id==='final-lift'&&x.state==='AT DELIVERY')&&r.w.player.grounded&&Math.abs(r.w.player.pos.y-4.265)<.06,20,'Separate final lift did not deliver actual capsule');assert.ok((r.ridingFrames.get('final-lift')??0)>240);assert.equal(report(r,'gallery-contact').status,'OBSERVED');assert.equal(report(r,'delivery-arrival').status,'OBSERVED');r.until(()=>journal(r).now.some(x=>x.id==='delivery-door'&&x.state==='OPEN'),2,'Actual final dock/pressure leaf never clears');r.walk(-1,-1);r.walk(-11,-1);r.finish('delivery-exit');
}
function lowerService(r:CompletionRoute){r.walk(5,13);r.until(()=>journal(r).now.some(x=>x.id==='vault-service-door'&&x.state==='OPEN'),2,'Actual lower SERVICE allocation never clears its door');r.walk(5,17);}
test('native C40 both requests retain identical momentum vault and high return remains back-facing from spawn',()=>{const a=new CompletionRoute(40),b=new CompletionRoute(40,true);try{assert.deepEqual(a.w.blueprint,b.w.blueprint);assert.deepEqual(journal(a).rows,[]);const face=a.w.surfaces.find(s=>s.id==='vault-return-floor');assert.ok(face);assert.ok(a.w.eye().clone().sub(face.center).dot(face.normal)<0,'Lower approach must be behind the upper floor face');assert.equal(a.pad('gallery-pressure').pressed,false);assert.equal(a.pad('service-pressure').pressed,false);}finally{a.w.dispose();b.w.dispose();}});
test('native C40 actual deep fling and independent cargo return still require the final gallery journey',()=>route(40,false,'fling-return-final-gallery',r=>{vaultLanding(r);returnVaultCargo(r);allocate40(r,'gallery-pressure');assert.equal(report(r,'gallery-contact').status,'OBSERVED');assert.equal(report(r,'delivery-arrival').status,'NOT YET OBSERVED');assert.ok(r.device('delivery-door').progress<.001);finalGallery(r);}));
test('native C40 same retrieval contrast delivers SERVICE and leaves final gallery unpowered',()=>route(40,true,'fling-return-lower-service',r=>{vaultLanding(r);returnVaultCargo(r);allocate40(r,'service-pressure');lowerService(r);r.finish('service-exit');assert.equal(r.pad('gallery-pressure').pressed,false);assert.equal(r.device('final-lift').moving,false);assert.equal(r.ridingFrames.get('final-lift')??0,0);assert.equal(report(r,'service-opening').status,'OBSERVED');}));
test('native C40 wrong completed SERVICE remains recoverable by actual GALLERY reallocation',()=>route(40,false,'wrong-service-reallocate-gallery',r=>{vaultLanding(r);returnVaultCargo(r);allocate40(r,'service-pressure');lowerService(r);assert.equal(r.w.campaignResult,undefined);assert.ok(r.events.some(e=>e.type==='wrong-exit'&&e.exit==='service-exit'));r.walk(5,13);r.walk(4.2,12);r.pickup('vault-cube');r.walk(3,12);allocate40(r,'gallery-pressure');assert.equal(r.pad('service-pressure').pressed,false);finalGallery(r);}));
test('native C40 a real plate departure leaves its true completed report but removes unfinished transfer power',()=>route(40,false,'current-pressure-brake-restore',r=>{vaultLanding(r);returnVaultCargo(r);allocate40(r,'gallery-pressure');const lift=r.device('final-lift');r.until(()=>lift.moving&&lift.position.y>.4&&lift.position.y<2.8,30,'No final movement to interrupt');r.pickup('vault-cube');r.walk(3,9);r.until(()=>!r.pad('gallery-pressure').pressed&&!lift.moving,1,'Actual returned-cargo removal did not stop current transfer');const stopped=lift.position.clone();r.tick(1);assert.ok(lift.position.distanceTo(stopped)<1e-5);assert.equal(report(r,'gallery-contact').status,'OBSERVED');assert.equal(report(r,'gallery-contact').text,'The GALLERY plate has been pressed. Check its current pressure.');assert.ok(journal(r).now.some(x=>x.id==='gallery-pressure'&&x.state==='EMPTY'));r.dropPad('vault-cube','gallery-pressure',1.8,9);finalGallery(r);}));
test('native C40 missed deep aperture reaches real basin then RETURN carries a reset-free recentered retry',()=>route(40,false,'actual-deep-miss-return-recenter',r=>{prepareDeep(r);r.walk(4.3,6.45);r.w.yaw=0;r.w.pitch=0;const crossings=r.w.player.crossings;for(let i=0;i<600&&!(r.w.player.grounded&&r.w.player.pos.y<-16);i++)r.step(forward);assert.equal(r.w.player.crossings,crossings);assert.ok(r.w.player.grounded&&Math.abs(r.w.player.pos.y+17.135)<.08,'Real miss did not land on deep floor');r.walk(4.3,-3.4);r.walk(4.3,-5.1);r.until(()=>r.w.player.grounded&&Math.abs(r.w.player.pos.y+21.135)<.08,4,'Actual bottom opening did not reach basin');r.walk(-1,10);r.walk(-5,10);r.control('vault-recovery');const d=r.device('vault-recovery');r.until(()=>!d.moving&&Math.abs(d.position.y+.2)<.015&&r.w.player.grounded&&Math.abs(r.w.player.pos.y-.865)<.06,12,'Actual RETURN deck failed loaded approach recovery');r.walk(-1,10);assert.equal(r.w.respawns,0);vaultLanding(r);returnVaultCargo(r);allocate40(r,'gallery-pressure');finalGallery(r);}));
}else test('unregistered C40 awaits explicit native acceptance',{skip:'Enable RIFT_COMPLETION_ROUTES=1 only in the assigned serial slot.'},()=>{});
